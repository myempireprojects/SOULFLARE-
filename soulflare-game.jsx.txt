const PR = {
  male:   { sub:"he",   obj:"him",  pos:"his" },
  female: { sub:"she",  obj:"her",  pos:"her" },
  other:  { sub:"they", obj:"them", pos:"their" },
};
function inj(t, p1, p2) {
  if (!t) return "";
  const a = PR[p1?.gender] || PR.other;
  const b = PR[p2?.gender] || PR.other;
  return t
    .replace(/{p1}/g,  p1?.name || "Partner 1")
    .replace(/{p2}/g,  p2?.name || "Partner 2")
    .replace(/{sub1}/g, a.sub).replace(/{obj1}/g, a.obj).replace(/{pos1}/g, a.pos)
    .replace(/{sub2}/g, b.sub).replace(/{obj2}/g, b.obj).replace(/{pos2}/g, b.pos)
    .replace(/{Sub1}/g, a.sub[0].toUpperCase() + a.sub.slice(1))
    .replace(/{Sub2}/g, b.sub[0].toUpperCase() + b.sub.slice(1));
}

function shuffle(arr){
  const a=[...arr];
  for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}
  return a;
}

            <button className="sfb" onClick={()=>setScreen("game")} style={{background:"none",color:"#6a5a7a",fontSize:14}}>← Back</button>
            <h2 style={{fontFamily:"'Playfair Display',serif",fontSize:24,color:"#d4a853",flex:1,textAlign:"center"}}>🃏 Cards of Fate</h2>
          </div>

          {/* Progress */}
          <div style={{display:"flex",alignItems:"center",gap:10,marginBottom:14}}>
            <span style={{color:"#5a4a6a",fontSize:12,whiteSpace:"nowrap"}}>{24-fateIdx} left</span>
            <div style={{height:4,flex:1,background:"#1a1030",borderRadius:2,overflow:"hidden"}}>
              <div style={{width:`${(fateIdx/24)*100}%`,height:"100%",background:"linear-gradient(90deg,#d4a853,#e8175d)",transition:"width .4s ease"}}/>
            </div>
            <span style={{color:"#d4a853",fontSize:12,fontWeight:700,whiteSpace:"nowrap"}}>{fateIdx}/24</span>
          </div>

          {!done?(
            <>
              {/* Turn indicator */}
              <div style={{background:"#110820",border:`1px solid ${fateTurn===0?"#e8175d33":"#d4a85333"}`,
                borderRadius:12,padding:"11px 16px",marginBottom:18,textAlign:"center"}}>
                <p style={{color:fateTurn===0?"#e8175d":"#d4a853",fontSize:17,fontWeight:700,fontFamily:"'Playfair Display',serif"}}>
                  {cur.name}'s turn to draw 🃏
                </p>
                <p style={{color:"#5a4a6a",fontSize:13,marginTop:3}}>Shuffle the deck in your hands, then tap to reveal</p>
              </div>

              {/* Card face-down / face-up */}
              <div style={{flex:1,display:"flex",alignItems:"center",justifyContent:"center"}}>
                {!fateFlip?(
                  // Face-down stack
                  <div style={{position:"relative",width:230,height:310,cursor:"pointer"}} onClick={()=>setFateFlip(true)}>
                    {[3,2,1].map(offset=>(
                      <div key={offset} style={{position:"absolute",top:offset*5,left:offset*4,right:-offset*4,
                        height:290,background:`#${offset===3?"09051a":offset===2?"0e0720":"130a28"}`,
                        border:"1px solid #1e1030",borderRadius:18,opacity:1-offset*.18}}/>
                    ))}
                    <div style={{position:"absolute",inset:0,
                      background:"linear-gradient(148deg,#1a0a38,#0c0520)",
                      border:"2px solid #d4a85344",borderRadius:18,
                      display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",
                      boxShadow:"0 0 48px #d4a85322",animation:"glow 2.8s ease-in-out infinite"}}>
                      <div style={{fontSize:62,marginBottom:14}}>🃏</div>
                      <p style={{fontFamily:"'Playfair Display',serif",fontSize:26,color:"#d4a853",fontWeight:900,letterSpacing:".08em"}}>FATE</p>
                      <p style={{color:"#5a4a6a",fontSize:14,marginTop:10}}>Tap to reveal your fate</p>
                    </div>
                  </div>
                ):(
                  // Revealed card
                  fc&&(
                    <div style={{width:"100%",maxWidth:370,
                      background:`linear-gradient(148deg,#150b28,#0e0620)`,
                      border:`2px solid ${tc[fc.type]||"#d4a853"}44`,borderRadius:20,padding:24,
                      boxShadow:`0 0 55px ${tc[fc.type]||"#d4a853"}18`,animation:"flipIn .5s ease"}}>
                      <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:16}}>
                        <span style={{background:tb[fc.type],border:`1px solid ${tc[fc.type]||"#d4a853"}44`,
                          color:tc[fc.type]||"#d4a853",padding:"5px 14px",borderRadius:20,fontSize:13,
                          textTransform:"uppercase",letterSpacing:".1em",fontWeight:700}}>{fc.type}</span>
                        <span style={{fontSize:42}}>{fc.emoji}</span>
                      </div>
                      <h3 style={{fontFamily:"'Playfair Display',serif",fontSize:24,color:"#f5e6d3",marginBottom:14,lineHeight:1.2}}>{fc.title}</h3>
                      <p style={{fontSize:17,lineHeight:1.8,color:"#d0bce8"}}>{fc.body}</p>
                      <div style={{marginTop:18,padding:"10px",background:"#09051a",borderRadius:10,textAlign:"center"}}>
                        <p style={{color:"#5a4a6a",fontSize:12}}>{cur.name}'s draw · Card {fateIdx+1} of 24</p>
                      </div>
                    </div>
                  )
                )}
              </div>

              {/* Action button */}
              <div style={{paddingTop:14,paddingBottom:4}}>
                {fateFlip?(
                  <button className="sfb" onClick={()=>{
                    if(fateIdx+1>=24){setFateIdx(24);}
                    else{setFateIdx(i=>i+1);setFateTurn(t=>(t+1)%2);setFateFlip(false);}
                  }} style={{width:"100%",padding:17,
                    background:"linear-gradient(135deg,#d4a853,#c49040)",borderRadius:14,color:"#06020e",
                    fontSize:18,fontFamily:"'Playfair Display',serif",fontWeight:700,boxShadow:"0 0 28px #d4a85344"}}>
                    {fateIdx+1>=24?"✨ All Cards Revealed!":"Done — Pass to "+next.name+" →"}
                  </button>
                ):(
                  <p style={{textAlign:"center",color:"#4a3a5a",fontSize:14,fontStyle:"italic",padding:16}}>Tap the card stack to draw…</p>
                )}
              </div>
            </>
          ):(
            // All 24 cards done
            <div style={{flex:1,display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",textAlign:"center",padding:20}}>
              <div style={{fontSize:72,marginBottom:16}}>🃏✨</div>
              <h3 style={{fontFamily:"'Playfair Display',serif",fontSize:28,color:"#d4a853",marginBottom:12}}>All Cards Revealed!</h3>
              <p style={{color:"#c4b0d8",fontSize:16,lineHeight:1.8,marginBottom:24}}>
                {p1.name} and {p2.name} have shared every card. What a journey.
              </p>
              <button className="sfb" onClick={()=>setScreen("game")} style={{padding:"16px 36px",
                background:"linear-gradient(135deg,#d4a853,#c49040)",borderRadius:14,color:"#06020e",
                fontSize:18,fontFamily:"'Playfair Display',serif",fontWeight:700}}>
                Return to Game 🔥
              </button>
            </div>
          )}
        </div>
      </div>
    );
  }

  // ══════════════════════════════════════════
  // FINALE
  // ══════════════════════════════════════════
  if(screen==="finale") return(
    <div style={root}><Sparks/>
      <div style={{position:"absolute",inset:0,background:"radial-gradient(ellipse at center,#e8175d12 0%,transparent 68%)",pointerEvents:"none"}}/>
      <div style={{textAlign:"center",zIndex:1,maxWidth:400,padding:"28px 20px",animation:"rise .8s ease",overflowY:"auto",maxHeight:"100vh"}}>
        <div style={{fontSize:78,animation:"hb 1.6s ease-in-out infinite",marginBottom:14}}>❤️‍🔥</div>
        <h2 style={{fontFamily:"'Playfair Display',serif",fontSize:"clamp(32px,9vw,50px)",
          background:"linear-gradient(135deg,#e8175d,#d4a853)",WebkitBackgroundClip:"text",WebkitTextFillColor:"transparent",marginBottom:8}}>
          SoulFlare Complete!</h2>
        {couples.length>0&&(
          <p style={{color:"#d4a853",fontSize:17,fontStyle:"italic",marginBottom:8}}>
            {couples.map(c=>`${c[0].name} & ${c[1].name}`).join(" · ")}</p>
        )}
        <HR/>

        {/* POINTS + AWARD */}
        <div style={{background:"#110820",border:"1px solid #d4a85333",borderRadius:16,padding:20,marginBottom:18}}>
          <p style={{color:"#d4a853",fontSize:24,fontWeight:700,fontFamily:"'Playfair Display',serif",marginBottom:8}}>💎 {points} SoulFlare Points</p>
          {points>=30?(
            <div style={{padding:"12px",background:"#e8175d14",border:"1px solid #e8175d44",borderRadius:12,animation:"pop .6s ease"}}>
              <p style={{color:"#e8175d",fontSize:20,fontWeight:700,fontFamily:"'Playfair Display',serif",marginBottom:4}}>🏆 SoulFlare Couple!</p>
              <p style={{color:"#c4b0d8",fontSize:14,lineHeight:1.6}}>You conquered every puzzle together. This title is yours.</p>
            </div>
          ):points>0?(
            <p style={{color:"#7a6a8a",fontSize:14,marginTop:4}}>Complete more puzzles next game to earn the SoulFlare Couple title!</p>
          ):(
            <p style={{color:"#5a4a6a",fontSize:14,marginTop:4}}>Look out for puzzle cards next time to earn points!</p>
          )}
        </div>

        <p style={{color:"#c4b0d8",fontSize:17,lineHeight:1.82,margin:"14px 0 20px",padding:"0 6px"}}>
          You have walked through truth, laughter, vulnerability, and love. May this flame burn brighter every day. 🙏
        </p>

        <div style={{background:"#110820",border:"1px solid #d4a85333",borderRadius:16,padding:"18px 20px",marginBottom:22,boxShadow:"0 0 32px #d4a85318"}}>
          <p style={{color:"#d4a853",fontSize:18,fontStyle:"italic",lineHeight:1.72,marginBottom:8}}>
            "And now these three remain: faith, hope and love. But the greatest of these is love."
          </p>
          <p style={{color:"#5a4a6a",fontSize:13}}>— 1 Corinthians 13:13</p>
        </div>

        <div style={{display:"flex",gap:12}}>
          <button className="sfb" onClick={()=>{
            clearG();setScreen("welcome");setRound(0);setMode("EASY");setUsedIds([]);setCard(null);setPoints(0);
            setPlayers([]);setCouples([]);couplesRef.current=[];
          }} style={{flex:1,padding:15,background:"#110820",border:"1px solid #1e1030",
            borderRadius:12,color:"#8a7a9a",fontSize:16,fontFamily:"'Playfair Display',serif"}}>🏠 Main Menu</button>
          <button className="sfb" onClick={()=>{
            clearG();const c=couplesRef.current;
            setRound(0);setMode("EASY");setUsedIds([]);setPoints(0);setCard(null);
            setScreen("transition");speak("Welcome back. Easy Mode begins.");
            setTimeout(()=>{setScreen("game");drawCard(0,c,[]);},3800);
          }} style={{flex:1,padding:15,background:"linear-gradient(135deg,#e8175d,#a8083e)",
            borderRadius:12,color:"#fff",fontSize:16,fontFamily:"'Playfair Display',serif",fontWeight:700}}>Play Again 🔥</button>
        </div>
      </div>
    </div>
  );

  return null;
}
